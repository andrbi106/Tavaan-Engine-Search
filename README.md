## استفاده

```php
use Tavaan\Engine\Search\SearchManager;
use Tavaan\Engine\Search\Drivers\NativeIndex;

// Manager با چند ایندکس
$m = new SearchManager(['driver' => 'native']);

// Index
$m->addDocument('products', 'p1', 'لپ تاپ ایسوس ROG');
$m->addDocument('products', 'p2', 'لپ تاپ لنوو ThinkPad');
$m->addDocument('products', 'p3', 'گوشی سامسونگ');

// Search
$results = $m->search('products', 'لپ تاپ', limit: 20);
// [['docId' => 'p1', 'score' => 1.5], ['docId' => 'p2', 'score' => 1.5]]

// Suggest
$suggestions = $m->suggest('products', 'لپ', limit: 5);

// Delete
$m->deleteDocument('products', 'p1');

// Count
$m->count('products');

// Direct driver
$idx = new NativeIndex();
$idx->index('d1', 'hello world');
$idx->search('hello');
```
